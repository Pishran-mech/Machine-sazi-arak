روش اتصال سایت به Google Sheets و ذخیره دائمی فیش‌ها

  1. وارد سایت sheets.google.com شوید و یک فایل جدید (Blank Spreadsheet) بسازید.
  2. از منوی بالا روی Extensions و سپس Apps Script کلیک کنید.
  3. کدهای قبلی را کاملاً پاک کرده و کد جدید زیر را کپی و جایگذاری کنید:
Google Apps Script (Code.gs) - نسخه پشتیبان فیش تصویری 
const HEADERS = ['id', 'trackingCode', 'fullName', 'major', 'entryYear', 'studentId', 'nationalId', 'mobile', 'emergencyPhone', 'lotteryStatus', 'reserveRank', 'paymentStatus', 'paidAmount', 'receiptRef', 'receiptName', 'createdAt', 'driveLink', 'receiptDataUrl'];

function getOrCreateSheet() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getActiveSheet();
  if (sheet.getLastRow() === 0) {
    sheet.appendRow(HEADERS);
  }
  return sheet;
}

function readAllFromSheet() {
  var sheet = getOrCreateSheet();
  var data = sheet.getDataRange().getValues();
  if (data.length <= 1) return [];
  var list = [];
  for (var r = 1; r < data.length; r++) {
    var row = data[r];
    if (!row[0] && !row[1]) continue;
    list.push({
      id: String(row[0] || ''),
      trackingCode: String(row[1] || ''),
      fullName: String(row[2] || ''),
      major: String(row[3] || ''),
      entryYear: String(row[4] || ''),
      studentId: String(row[5] || ''),
      nationalId: String(row[6] || ''),
      mobile: String(row[7] || ''),
      emergencyPhone: String(row[8] || ''),
      lotteryStatus: String(row[9] || 'در انتظار قرعه‌کشی'),
      reserveRank: row[10] === '-' || row[10] === '' ? null : Number(row[10]),
      paymentStatus: String(row[11] || 'پس از قرعه‌کشی'),
      paidAmount: Number(row[12] || 0),
      receiptRef: String(row[13] || ''),
      receiptName: String(row[14] || ''),
      createdAt: String(row[15] || ''),
      driveLink: String(row[16] || ''),
      receiptDataUrl: String(row[17] || '')
    });
  }
  return list;
}

function saveReceiptToDriveIfPossible(reg) {
  try {
    if (!reg.receiptDataUrl || reg.receiptDataUrl.indexOf('base64,') === -1) return reg.driveLink || '';
    var folders = DriveApp.getFoldersByName('Pishran_Arak_Receipts');
    var folder = folders.hasNext() ? folders.next() : DriveApp.createFolder('Pishran_Arak_Receipts');
    var base64Data = reg.receiptDataUrl.split('base64,')[1];
    var blob = Utilities.newBlob(Utilities.base64Decode(base64Data), 'image/jpeg', (reg.studentId || reg.trackingCode) + '_' + (reg.fullName || 'receipt') + '.jpg');
    var file = folder.createFile(blob);
    return file.getUrl();
  } catch (err) {
    return reg.driveLink || '';
  }
}

function writeAllToSheet(list) {
  var sheet = getOrCreateSheet();
  sheet.clearContents();
  sheet.appendRow(HEADERS);
  var rows = list.map(function(r) {
    return [
      r.id || '', r.trackingCode || '', r.fullName || '', r.major || '', r.entryYear || '',
      r.studentId || '', r.nationalId || '', r.mobile || '', r.emergencyPhone || '',
      r.lotteryStatus || '', r.reserveRank || '-', r.paymentStatus || '', r.paidAmount || 0,
      r.receiptRef || '', r.receiptName || '', r.createdAt || '', r.driveLink || '', r.receiptDataUrl || ''
    ];
  });
  if (rows.length > 0) {
    sheet.getRange(2, 1, rows.length, HEADERS.length).setValues(rows);
  }
}

function doGet() {
  var list = readAllFromSheet();
  return ContentService.createTextOutput(JSON.stringify(list)).setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  var lock = LockService.getScriptLock();
  lock.waitLock(10000);
  try {
    var body = JSON.parse(e.postData.contents);
    var list = readAllFromSheet();

    if (body.action === 'REGISTER' && body.payload) {
      list.unshift(body.payload);
    } else if (body.action === 'UPLOAD_RECEIPT' && body.payload) {
      var p = body.payload;
      p.driveLink = saveReceiptToDriveIfPossible(p);
      var found = false;
      for (var i = 0; i < list.length; i++) {
        if (list[i].id === p.id || list[i].trackingCode === p.trackingCode || list[i].studentId === p.studentId) {
          list[i].paymentStatus = p.paymentStatus;
          list[i].paidAmount = p.paidAmount;
          list[i].receiptName = p.receiptName;
          list[i].receiptRef = p.receiptRef || '';
          list[i].receiptDataUrl = p.receiptDataUrl || list[i].receiptDataUrl || '';
          list[i].driveLink = p.driveLink || list[i].driveLink || '';
          found = true;
          break;
        }
      }
      if (!found) list.unshift(p);
    } else if (body.allRegistrations) {
      var oldMap = {};
      list.forEach(function(item) { oldMap[item.id] = item; });
      list = body.allRegistrations.map(function(item) {
        var prev = oldMap[item.id] || {};
        if (!item.receiptDataUrl && prev.receiptDataUrl) item.receiptDataUrl = prev.receiptDataUrl;
        if (!item.driveLink && prev.driveLink) item.driveLink = prev.driveLink;
        return item;
      });
    }

    writeAllToSheet(list);
    return ContentService.createTextOutput(JSON.stringify({status: 'ok'})).setMimeType(ContentService.MimeType.JSON);
  } finally {
    lock.releaseLock();
  }
}

  3. لینک Web App URL را کپی کرده و در بخش «تنظیمات» پنل برگزارکننده قرار دهید.
